Sets up i18next to load translation files over HTTP and swap languages at runtime. applyTranslations() fills in any element
tagged with data-i18n.
frontend/src/js/i18n.js
i18next.use(i18nextHttpBackend);
function initI18n(callback) {
  i18next.init(
    {
      lng: localStorage.getItem('language') || 'en',
      fallbackLng: 'en',
      debug: false,
      backend: {
        loadPath: '/public/locales/{{lng}}/translation.json',
      },
    },
    () => {
      applyTranslations();
      if (callback) callback();
    }
  );
}
function applyTranslations() {
  document.querySelectorAll('[data-i18n]').forEach((el) => {
    const key = el.getAttribute('data-i18n');
    el.textContent = i18next.t(key);
  });
  document.querySelectorAll('[data-i18n-placeholder]').forEach((el) => {
    const key = el.getAttribute('data-i18n-placeholder');
    el.placeholder = i18next.t(key);
  });
}
function changeLanguage(lng) {
  i18next.changeLanguage(lng, () => {
    localStorage.setItem('language', lng);
    applyTranslations();
  });
}

frontend/src/js/components/languageSwitcher.js
function renderLanguageSwitcher() {
  const container = document.createElement('div');
  container.innerHTML = `
    
      English
      IIII
    
  `;
  const select = container.querySelector('#language-select');
  select.value = localStorage.getItem('language') || 'en';
  select.addEventListener('change', (e) => {
    changeLanguage(e.target.value);
  });
  return container;
}
